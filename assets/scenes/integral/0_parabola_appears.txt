# Scene 0: The Parabola Appears
# f(x) = x² drawn as a parametric curve. It scales in from nothing.

# x axis
line x_axis  -0.5 0 0   2.5 0 0   0.5 0.5 0.5

# y axis
line y_axis  0 -0.5 0   0 4.5 0   0.5 0.5 0.5

# f(x) = x^2, parametric: x=t, y=t^2, z=0, t in [0, 2]
curve parabola  t  t^2  0   0.2 0.9 0.4  0 2

set parabola scale 0 0 0

wait 1.0
animate parabola scale ease_out path t t t 2.0
