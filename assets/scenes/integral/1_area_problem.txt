# Scene 1: The Area Problem
# We want to find the area under f(x)=x² from 0 to 2.
# A shaded region (quad) hints at the area. Boundary lines mark [0,2].

# Axes
line x_axis  -0.5 0 0   2.5 0 0   0.5 0.5 0.5
line y_axis  0 -0.5 0   0 4.5 0   0.5 0.5 0.5

# f(x) = x^2
curve parabola  t  t^2  0   0.2 0.9 0.4  0 2

# Boundary lines at x=0 and x=2 (vertical dashed)
line bound_left   0 0 0   0 0.1 0   1 0.8 0.2
line bound_right  2 0 0   2 4 0     1 0.8 0.2

# A rough triangular quad hinting at the shaded area
quad area_hint yellow.png  0 0 0  2 0 0  2 4 0  0 0 0

set area_hint scale 0 0 0
set bound_left scale 0 0 0
set bound_right scale 0 0 0

wait 1.0
animate bound_left scale ease_out path t t t 1.0
animate bound_right scale ease_out path t t t 1.0

wait 1.5
animate area_hint scale ease_out path t t t 1.5
