# =============================================================
#  scene0 — "El pueblo al amanecer"  (~20 s)
#  Un pequeño pueblo: suelo, un camino, dos casas, un molino,
#  una estrella que se apaga y un pajaro que sobrevuela.
#
#  Contiene al menos 1 de cada entidad:
#    quad, line, curve, mesh, group, preset
#  y las 4 formas de animacion:
#    set (instantanea), linear, smooth, path
# =============================================================

# --- Fondo (papel/cielo) -------------------------------------
background backgrounds/paper.png

# --- Suelo del pueblo (quad de madera, 20x20) ----------------
quad ground wood  20 20
# el suelo empieza plano en el piso, mirando hacia arriba
set ground rotation -90 0 0
set ground position 0 -0.5 0

# --- Camino de tierra que cruza el pueblo (line con flecha) --
line road  -8 -0.49 3   8 -0.49 3   0.6 0.45 0.3  arrow_default

# --- Casas del pueblo (mesh cubo con textura de madera) ------
mesh house_a  cube.obj  wood  1.5
mesh house_b  cube.obj  wood  1.2

# --- Techo de cada casa (quad) para agruparlos ---------------
quad roof_a  yellow  1.6 1.6
quad roof_b  yellow  1.3 1.3

# --- Molino: aspas hechas con una curva parametrica ----------
# circulo de radio ~1.2 => aspas girando
curve blades  1.2*cos(t)  1.2*sin(t)  0   0.9 0.9 0.95   0 6.2832

# --- Estrella decorativa (preset de lineas) ------------------
preset morning_star star.txt

# =============================================================
#  Posiciones iniciales (todas con `set` = animacion instantanea)
# =============================================================
set house_a position -3 0 0
set house_b position  3 0 -1

set roof_a position -3 1.1 0
set roof_b position  3 1.0 -1

set blades position 0 2.2 -4
set morning_star position 0 4 -3
set morning_star scale 1 1 1

# --- Agrupar casa_a con su techo (group) ---------------------
group homestead house_a roof_a

# =============================================================
#  TIMELINE  (0 s -> 20 s)
# =============================================================

# t=0 : sale el sol -> la estrella se encoge suavemente (smooth)
animate morning_star scale ease_out smooth  1 1 1   0.05 0.05 0.05   4.0

# t=0 : las aspas del molino giran continuamente (linear rotation)
animate blades rotation linear linear  0 0 0   0 0 360   6.0

# t=0 : el conjunto casa+techo se levanta (linear) para "construirse"
animate homestead position linear linear  -3 -2 0   -3 0 0   3.0

wait 4

# t=4 : la casa_b cambia de color al iluminarse (smooth color)
animate house_b color ease_in_out smooth  1 1 1   1.0 0.8 0.4   3.0

# t=4 : las aspas siguen girando otra vuelta mas rapido (linear)
animate blades rotation linear linear  0 0 360   0 0 720   4.0

wait 4

# t=8 : un pajaro (reutilizamos roof_b como "pajaro" simbolico)
#       vuela describiendo una trayectoria en 8 (path).
#       En modo path, t va de 0 a 1 durante la animacion; por eso
#       multiplicamos por 6.2832 (2*pi) para recorrer una vuelta completa.
animate roof_b position linear path  4*sin(6.2832*t)  cos(12.566*t)  4*sin(3.1416*t)  8.0

# t=8 : el conjunto homestead da un pequeno salto (smooth)
animate homestead position ease_in_out smooth  -3 0 0   -3 0.6 0   -3 0 0   4.0

wait 6

# t=14 : el molino se aleja hacia el fondo (linear position)
animate blades position linear linear  0 2.2 -4   0 2.2 -9   4.0

# t=14 : la casa_b vuelve a su color original (smooth)
animate house_b color ease_out smooth  1.0 0.8 0.4   1 1 1   4.0

wait 6

# t=20 : fin de la escena
