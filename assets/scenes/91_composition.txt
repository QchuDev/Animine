# Scene 10: Composition — sin(t) + cos(t)
# Adding two waves creates a new wave with different amplitude and phase.
# sin(t) + cos(t) = √2 * sin(t + π/4)

line x_axis  -1 0 0   7 0 0   0.3 0.3 0.3
line y_axis  0 -2 0   0 2 0   0.3 0.3 0.3

# Individual waves (faded)
curve sine    t/(2*pi)*4+1  sin(t)      0   0.2 0.5 0.2  -12.57 12.57
curve cosine  t/(2*pi)*4+1  cos(t)      0   0.5 0.2 0.2  -12.57 12.57

# The composed wave: sin(t) + cos(t)
curve composed  t/(2*pi)*4+1  sin(t)+cos(t)  0   1.0 0.9 0.2  -12.57 12.57

# Dots on each wave
quad sin_dot yellow.png 0.06 0.06
quad cos_dot yellow.png 0.06 0.06
quad sum_dot yellow.png 0.12 0.12

# Sin dot
animate sin_dot position linear path t*4+1 sin(t*6.28) 0 8.0

# Cos dot
animate cos_dot position linear path t*4+1 cos(t*6.28) 0 8.0

# Sum dot — the composition
animate sum_dot position linear path t*4+1 sin(t*6.28)+cos(t*6.28) 0 8.0
