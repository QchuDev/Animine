# Scene 3: The X Projection — Cosine
# cos(θ) is the X component. A vertical line shows the dot's X value.

curve circle  cos(t)  sin(t)  0   0.5 0.5 0.5

# The dot on the circle
quad dot yellow.png 0.1 0.1

# A projection marker on the X axis (red = cosine)
quad x_proj yellow.png 0.08 0.08

# Vertical line from X axis to dot (defined 0,0,0 → 0,1,0, scaled in Y)
line v_line  0 0 0   0 1 0   0.9 0.2 0.2

set dot position 0 0 0
set x_proj position 0 0 0
set v_line position 0 0 0

# Dot traces circle
animate dot position linear path cos(t*6.28) sin(t*6.28) 0 8.0

# X projection follows only the X component
animate x_proj position linear path cos(t*6.28) 0 0 8.0

# v_line: position tracks X, scale in Y stretches to reach the dot
animate v_line position linear path cos(t*6.28) 0 0 8.0
animate v_line scale linear path 0 sin(t*6.28)-1 0 8.0
