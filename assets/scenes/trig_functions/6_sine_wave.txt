# Scene 6: The Sine Wave
# sin(θ) is the Y component. The wave unfolds to the right along the X axis.

curve circle  cos(t)  sin(t)  0   0.4 0.4 0.4

# The sine wave (unfolds rightward: x = t scaled, y = sin(t))
curve sine  t/(2*pi)*4  sin(t)  0   0.2 0.9 0.3  -12.57 12.57

# Dot on circle
quad dot yellow.png 0.1 0.1

# Dot tracing the sine wave
quad wave_dot yellow.png 0.08 0.08

set dot position 0 0 0
set wave_dot position 0 0 0

# Circle dot traces
animate dot position linear path cos(t*6.28) sin(t*6.28) 0 8.0

# Wave dot: x advances rightward, y = sin(θ)
animate wave_dot position linear path t*4 sin(t*6.28) 0 8.0
