# Helix — 3D spiral combining circular motion with linear advance
# Demonstrates parametric curves in 3D space

# The helix: x=cos(t), y=sin(t), z=t scaled
curve helix  cos(t)  sin(t)  t/3-1   0.3 0.7 1.0

# Its projection on the XY plane (circle)
curve xy_proj  cos(t)  sin(t)  -1   0.5 0.5 0.5

# Its projection on the XZ plane (cosine wave)
curve xz_proj  cos(t)  -1.5  t/3-1   0.5 0.5 0.5

# Reference axes
line x_axis  -2 0 0   2 0 0   1 0.3 0.3
line y_axis  0 -2 0   0 2 0   0.3 1 0.3
line z_axis  0 0 -2   0 0 2   0.3 0.3 1

# A point climbing the helix (one full turn, z from -1 to 1.4)
quad climber  yellow.png  0.1 0.1

# Helix: x=cos(θ), y=sin(θ), z advances linearly. θ goes 0→2π for one turn.
animate climber  position  linear  path  cos(t*6.28)  sin(t*6.28)  t*2.4-1   10.0
