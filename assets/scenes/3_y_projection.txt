# Scene 3: The Y Projection — Sine
# As the dot moves around the circle, a horizontal line shows its Y value.

line x_axis  -2 0 0   2 0 0   0.3 0.3 0.3
line y_axis  0 -2 0   0 2 0   0.3 0.3 0.3
curve circle  cos(t)  sin(t)  0   0.5 0.5 0.5

# The dot on the circle
quad dot yellow.png 0.1 0.1

# A projection marker on the Y axis (green = sine)
quad y_proj yellow.png 0.08 0.08

# Horizontal line from Y axis to dot (defined 0,0,0 → 1,0,0, scaled in X)
line h_line  0 0 0   1 0 0   0.2 0.8 0.2

# Dot traces circle
animate dot position linear path cos(t*6.28) sin(t*6.28) 0 8.0

# Y projection follows only the Y component
animate y_proj position linear path 0 sin(t*6.28) 0 8.0

# h_line: position tracks Y, scale in X stretches to reach the dot
animate h_line position linear path 0 sin(t*6.28) 0 8.0
animate h_line scale linear path cos(t*6.28)-1 0 0 8.0
