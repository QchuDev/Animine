// En C/C++ tenemos que hacernos cargo de los espacios de memoria que ocupamos
// no hay garbage collector

char* c = new char(); // addr: 0xf1
*c = 'b'; // addr: 0xf1 val: 'b'                

c = new char(); // addr: 0xf3           <--- MEMORY LEAK (no libera 0xf1 antes de usar otro espacio de memoria)
*c = 'g': // addr: 0xf3 val: 0xf3
delete c; // addr: 0xf3 liberado
